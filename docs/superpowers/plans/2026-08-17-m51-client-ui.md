> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Finish the M51 user-facing surface: grade change + share on web pebble detail, a `/connections/[id]` shared-pebbles page, and grade selector + badge + share sheet on iOS and Android.

**Architecture:** Three independent PRs (web #711 → iOS #709 → Android #710), zero database changes — everything rides on the #712 RLS/RPCs. Web extracts a shared `VisibilityMenu` consumed by both the composer chip and the detail badge; the connection page uses direct client reads (`connections` row → peer id → `pebbles` by `user_id`, RLS-trimmed). Natives add a `Menu`/`DropdownMenu` chip bound to the existing draft `visibility`, per-grade badge icons, and a system share sheet for public pebbles at `https://www.pbbls.app/p/<id>`.

**Tech Stack:** Next.js 16 + shadcn (web), SwiftUI/iOS 17 + Swift Testing (iOS), Compose + JUnit4 (Android, CI-only builds).

**Spec:** `docs/superpowers/specs/2026-08-17-m51-client-ui-design.md` (D1–D8).

- TypeScript strict, no `any`, no type assertions unless unavoidable.
- One logical change per commit, conventional-commit messages, lowercase, no period.

- [ ] **Step 2: Implement `peerIdOf`**

`apps/web/lib/data/connection-peer.ts`:

```ts
/**
 * The peer in a canonical connection row (M49: one ordered row, user_a <
 * user_b). RLS already scopes the row to its two members, so a non-member
 * result can only mean a stale session — callers treat null as "not mine".
 */
export function peerIdOf(
  row: { user_a: string; user_b: string },
  selfId: string,
): string | null {
  if (row.user_a === selfId) return row.user_b
  if (row.user_b === selfId) return row.user_a
  return null
}
```

```ts
  async listConnectionSharedPebbles(
    connectionId: string,
  ): Promise<SharedConnectionPebble[] | null> {
    const { data: row, error: rowError } = await this.supabase
      .from("connections")
      .select("id, user_a, user_b")
      .eq("id", connectionId)
      .maybeSingle()
    if (rowError) throw new Error(`connection lookup failed: ${rowError.message}`)
    if (!row) return null

    const peerId = peerIdOf(row, this.userId)
    if (!peerId) return null

    // The widened pebbles_select (M51) trims this to 'private' + 'public'
    // rows; emotions is reference data (RLS `using (true)`), so the embed
    // cannot blank rows.
    const { data, error } = await this.supabase
      .from("pebbles")
      .select("id, name, happened_at, visibility, render_svg, emotions(id, slug, name, color)")
      .eq("user_id", peerId)
      .order("happened_at", { ascending: false })
    if (error) throw new Error(`shared pebbles fetch failed: ${error.message}`)

    return (data ?? []).flatMap((p) => {
      // The FK embed is typed as an object by PostgREST for a to-one relation;
      // drop rows with a broken embed rather than rendering an empty emotion.
      const emotion = p.emotions
      if (!emotion?.id || !emotion.slug || !emotion.name || !emotion.color) return []
      return [{
        id: p.id,
        name: p.name,
        happened_at: p.happened_at,
        visibility: p.visibility as SharedConnectionPebble["visibility"],
        emotion: { id: emotion.id, slug: emotion.slug, name: emotion.name, color: emotion.color },
        render_svg: p.render_svg,
      }]
    })
  }
```

- [ ] **Step 2: The hook** — `apps/web/lib/data/useConnectionPebbles.ts` (mirror `useConnections`' provider/cancelled/defer pattern exactly, including the `await Promise.resolve()` deferral comment):

```ts
"use client"

import { useEffect, useState } from "react"
import { useDataProvider } from "@/lib/data/provider-context"
import type { SharedConnectionPebble } from "@/lib/types"

/**
 * The pebbles one connection shares with the viewer (M51). Fetched on screen
 * open like the connections list; `notFound` covers both a foreign and a
 * removed connection id (indistinguishable by design).
 */
export function useConnectionPebbles(connectionId: string) {
  const { provider } = useDataProvider()
  const [pebbles, setPebbles] = useState<SharedConnectionPebble[]>([])
  const [notFound, setNotFound] = useState(false)
  const [loading, setLoading] = useState(() => provider !== null)
  const [error, setError] = useState<Error | null>(null)

  useEffect(() => {
    if (!provider) return
    let cancelled = false
    void (async () => {
      await Promise.resolve()
      if (cancelled) return
      setLoading(true)
      setError(null)
      try {
        const rows = await provider.listConnectionSharedPebbles(connectionId)
        if (cancelled) return
        if (rows === null) setNotFound(true)
        else setPebbles(rows)
        setLoading(false)
      } catch (err) {
        if (cancelled) return
        console.error("[connection-pebbles] load failed", err)
        setError(err instanceof Error ? err : new Error("Failed to load shared pebbles"))
        setLoading(false)
      }
    })()
    return () => {
      cancelled = true
    }
  }, [provider, connectionId])

  return { pebbles, notFound, loading, error }
}
```

fun VisibilityChip(

    onChange: (Visibility) -> Unit,
    modifier: Modifier = Modifier,

    Box(modifier) {

            leadingIcon = {
                Icon(

                    contentDescription = null,
                    modifier = Modifier.size(16.dp),
                )
            },
            label = { PebblesText(stringResource(value.labelRes), PebblesTypography.caption) },
        )
        DropdownMenu(expanded = expanded, onDismissRequest = { expanded = false }) {
            Visibility.entries.forEach { grade ->
                DropdownMenuItem(
                    leadingIcon = {
                        Icon(
                            painter = painterResource(grade.iconRes),
                            contentDescription = null,
                            modifier = Modifier.size(16.dp),
                        )
                    },
                    text = { PebblesText(stringResource(grade.labelRes), PebblesTypography.body) },
                    onClick = {
                        onChange(grade)
                        expanded = false
                    },
                )
            }
        }
    }
}
